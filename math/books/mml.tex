\section{Linear Algebra}

\subsection{vectors and systems}
\begin{itemize}
  \item Polynomials form a vector space.

  \item Elements of $\mathbb{R}^n$ (tuples of $n$ real numbers) are vectors.
  For example,
    \[
      a = \begin{bmatrix}
        1\\2\\3
      \end{bmatrix} \in \mathbb{R}^3.
    \]

  \item Adding two vectors $a,b \in \mathbb{R}^n$ component-wise results
  in another vector: $a+b=c \in \mathbb{R}^n$.

  \item Scaling: $a \in \mathbb{R}^n,\ \lambda \in \mathbb{R}
  \Longrightarrow \lambda a \in \mathbb{R}^n$.

  \item Systems of equations: solution by contradiction,
    \begin{align}
      x_1 + x_2 + x_3 &= 3 \tag{1}\\
      x_1 - x_2 + 2x_3 &= 2 \tag{2}\\
      2x_1 + 3x_3 &= 1 \tag{3}
    \end{align}
  has \emph{no solution}: Adding the first two equations yields
  $2x_1 + 3x_3 = 5$, which contradicts the third equation (3).

  \item Systems of equations: idk this is kinda basic but still,
    \begin{align}
      x_1 + x_2 + x_3 &= 3 \tag{1}\\
      x_1 - x_2 + 2x_3 &= 2 \tag{2}\\
      x_2 + x_3 &= 2 \tag{3}
    \end{align}
  (1) + (2) = $2x_1 + 3x_3 = 5$, i.e., $x_3 = 1$.

  From (3): we then get that $x_2 = 1$. so $(1,1,1)$ is the
  only possible and \emph{unique solution}

  \item Systems of equations: omition
    \begin{align}
      x_1 + x_2 + x_3 &= 3 \tag{1}\\
      x_1 - x_2 + 2x_3 &= 2 \tag{2}\\
      2x_1 + 3x_3 &= 5 \tag{3}
    \end{align}
    Since (1) + (2) = (3), we omit the third equation. solve with (1) and (2)
    and let $x_3 = a \in \mathbb{R}$.

  \item Solving systems with vectors: let $a_{ij} \in \mathbb{R}$.
  \[
    \begin{bmatrix}
      a_{11} \\
      \vdots \\
      a_{m1}
    \end{bmatrix} x_1 + \begin{bmatrix}
      a_{12} \\
      \vdots \\
      a_{m2}
    \end{bmatrix} x_2 + \cdots + \begin{bmatrix}
      a_{1n} \\
      \vdots \\
      a_{mn}
    \end{bmatrix} x_n = \begin{bmatrix}
      b_1 \\
      \vdots \\
      b_m
    \end{bmatrix}
  \]
  \[
    \Rightarrow\quad
    \begin{bmatrix}
      a_{11} & \cdots & a_{1n} \\
      \vdots &        & \vdots \\
      a_{m1} & \cdots & a_{mn}
    \end{bmatrix}
    \begin{bmatrix}
      x_1 \\
      \vdots \\
      x_n
    \end{bmatrix} = \begin{bmatrix}
      b_1 \\
      \vdots \\
      b_m
    \end{bmatrix}
  \]

\end{itemize}

\subsection{matrices}

\subsubsection{Matrix Addition and Multiplication}

Sum of Satrices $A, B \in \mathbb{R}^{m \times n}$ is defined as
the element-wise sum, i.e.,
\[
  A + B := \begin{bmatrix}
    a_{11} + b_{11} & \cdots & a_{1n} + b_{1n} \\
    \vdots & & \vdots \\
    a_{m1} + b_{m1} & \cdots & a_{mn} + b_{mn}
  \end{bmatrix} \in \mathbb{R}^{m \times n}.
  \tag{2.12}
\]

For matrices $A \in \mathbb{R}^{m \times n}$ and
$B \in \mathbb{R}^{n \times k}$, the elements $c_{ij}$ of the product
$C = AB \in \mathbb{R}^{m \times k}$ are computed as
\[
  c_{ij} = \sum_{l=1}^{n} a_{il}b_{lj},
  \qquad i = 1, \ldots, m, \quad j = 1, \ldots, k.
  \tag{2.13}
\]
This is the usual matrix multiplication.
